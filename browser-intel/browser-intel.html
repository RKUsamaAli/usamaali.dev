How to use in your application

Integration Example (JavaScript) 
// 1. Include the SDK script in your HTML
// <script src="https://usamaali.dev/browser-intel/browser-intel-sdk.minify.v1.js"></script>

// 2. Initialize and get the fingerprint
(async () => {
    // Initialize the SDK
    const sdk = new BrowserIntelSDK();
    await sdk.init();

    // Specify stability threshold (0-100)
    // Lower means more stable/less unique, higher means less stable/more unique
    const thresholdValue = 20; 
    
    // Generate fingerprint
    const result = await sdk.getFingerprint(thresholdValue);

    // Access the generated hash
    console.log("Device Fingerprint:", result.hash);
    
    // Access the raw data payload
    console.log("Raw Payload:", result.payload);
})();
